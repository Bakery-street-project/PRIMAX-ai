     cd /home/boozelee/Desktop/superbrain-x

     cat <<'EOF' > superbrain_gemini_analyzer.py
     #!/usr/bin/env python3
     import google.generativeai as genai
     from google.oauth2 import service_account
     from google.auth.transport.requests import Request
     import os

     creds = service_account.Credentials.from_service_account_file(
         '/home/boozelee/.config/gcloud/application_default_credentials.json',
         scopes=['https://www.googleapis.com/auth/generative-language']
     )
     creds.refresh(Request())
     genai.configure(transport='rest', credentials=creds)
     model = genai.GenerativeModel('gemini-1.5-pro')
     response = model.generate_content("Analyze Superbrain 2.0 for breakthroughs.")
     print(response.text)
     EOF
     chmod +x superbrain_gemini_analyzer.py

     EOF
